    // 使用 RD_SYS_T 指令读取系统时间
    #RD_Status := RD_SYS_T(#System_DateTime);
    
    // 检查状态码是否为成功
    IF #RD_Status = 0 THEN
        // 提取日期部分并更新当前日期输出
        #Current_Date_Output := DT_TO_DATE(#System_DateTime);
        
        // 动态生成密码
        #Temp_Password := #Base_Temp_Code XOR (DATE_TO_DWORD(#Current_Date_Output) * #Run_Duration_Days);
        #Perm_Password := #Base_Perm_Code XOR DATE_TO_DWORD(#Current_Date_Output);
        
        // 调试输出
        #Debug_Temp_Password := #Temp_Password;
        #Debug_Perm_Password := #Perm_Password;
        
        // 只有当密码写入标志为 1 时，进行密码验证
        IF #Password_Write_Enable THEN
            // 验证输入密码
            IF #Input_Password = #Temp_Password THEN
                #Temporary_Unlock := TRUE;
                #Stored_Date := #Current_Date_Output;  // 更新初始日期
                #Password_Error := FALSE;  // 清除错误标志
                #Password_Error_Timer := T#0S; // 清除计时器
            ELSIF #Input_Password = #Perm_Password THEN
                #Permanent_Unlock := TRUE;
                #Stored_Date := #Current_Date_Output;  // 更新初始日期
                #Password_Error := FALSE;  // 清除错误标志
                #Password_Error_Timer := T#0S; // 清除计时器
            ELSE
                // 如果密码错误且错误位未激活，触发错误标志
                IF NOT #Password_Error THEN
                    #Password_Error := TRUE;
                    #Password_Error_Timer := T#2S;  // 设置2秒计时器
                END_IF;
            END_IF;
        END_IF;
        
        // 密码错误标志计时器逻辑
        IF #Password_Error AND #Password_Error_Timer > T#0S THEN
            #Password_Error_Timer := #Password_Error_Timer - T#10MS; // 假设周期为10ms
            IF #Password_Error_Timer <= T#0S THEN
                #Password_Error := FALSE; // 复位错误标志
            END_IF;
        END_IF;
        
        // 锁定和解锁逻辑
        IF #Permanent_Unlock THEN
            #PLC_Lock := FALSE;
            #WICC_Lock := FALSE;
        ELSE
            IF DATE_TO_DWORD(#Current_Date_Output) - DATE_TO_DWORD(#Stored_Date) <= #Run_Duration_Days THEN
                #PLC_Lock := FALSE;
                #WICC_Lock := FALSE;
            ELSE
                #Temporary_Unlock := FALSE;
                #PLC_Lock := TRUE;
                #WICC_Lock := TRUE;
            END_IF;
        END_IF;
        
        // 时间异常检测
        IF DATE_TO_DWORD(#Current_Date_Output) < DATE_TO_DWORD(#Stored_Date) THEN
            #Error_Flag := TRUE;
            #PLC_Lock := TRUE;
            #WICC_Lock := TRUE;
        ELSE
            #Error_Flag := FALSE;
        END_IF;
        
    ELSE
        // 如果读取失败，可以根据需要处理，例如：触发故障标志
        #Error_Flag := TRUE;
    END_IF;
