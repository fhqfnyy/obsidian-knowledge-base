## 二、完整 VBA 宏代码
```
Sub ColorByLogic_AM_AN_AO_AP()

    Dim r As Long
    Dim lastRow As Long
    
    ' 从第3行开始，按 X 列确定数据行数
    lastRow = Cells(Rows.Count, "X").End(xlUp).Row

    For r = 3 To lastRow
        
        ' ===== AM 列（X / AE / AK 综合）=====
        With Cells(r, "AM")
            .Interior.Pattern = xlSolid
            
            If Cells(r, "X").Value > 0.15 _
            Or Cells(r, "AE").Value < -0.15 _
            Or Cells(r, "AK").Value < -0.15 Then
                
                .Interior.Color = RGB(255, 0, 0)        ' 红
                
            ElseIf Cells(r, "X").Value > 0.1 _
            Or Cells(r, "AE").Value < -0.1 _
            Or Cells(r, "AK").Value < -0.1 Then
                
                .Interior.Color = RGB(255, 165, 0)      ' 橙
                
            ElseIf Cells(r, "X").Value > 0.05 _
            Or Cells(r, "AE").Value < -0.05 _
            Or Cells(r, "AK").Value < -0.05 Then
                
                .Interior.Color = RGB(255, 255, 0)      ' 黄
                
            Else
                .Interior.Color = RGB(0, 176, 80)       ' 绿
            End If
        End With
        
        ' ===== AN 列（只看 X）=====
        With Cells(r, "AN")
            .Interior.Pattern = xlSolid
            
            If Cells(r, "X").Value > 0.15 Then
                .Interior.Color = RGB(255, 0, 0)
                
            ElseIf Cells(r, "X").Value > 0.1 Then
                .Interior.Color = RGB(255, 165, 0)
                
            ElseIf Cells(r, "X").Value > 0.05 Then
                .Interior.Color = RGB(255, 255, 0)
                
            Else
                .Interior.Color = RGB(0, 176, 80)
            End If
        End With
        
        ' ===== AO 列（只看 AE）=====
        With Cells(r, "AO")
            .Interior.Pattern = xlSolid
            
            If Cells(r, "AE").Value < -0.15 Then
                .Interior.Color = RGB(255, 0, 0)
                
            ElseIf Cells(r, "AE").Value < -0.1 Then
                .Interior.Color = RGB(255, 165, 0)
                
            ElseIf Cells(r, "AE").Value < -0.05 Then
                .Interior.Color = RGB(255, 255, 0)
                
            Else
                .Interior.Color = RGB(0, 176, 80)
            End If
        End With
        
        ' ===== AP 列（只看 AK）=====
        With Cells(r, "AP")
            .Interior.Pattern = xlSolid
            
            If Cells(r, "AK").Value < -0.15 Then
                .Interior.Color = RGB(255, 0, 0)
                
            ElseIf Cells(r, "AK").Value < -0.1 Then
                .Interior.Color = RGB(255, 165, 0)
                
            ElseIf Cells(r, "AK").Value < -0.05 Then
                .Interior.Color = RGB(255, 255, 0)
                
            Else
                .Interior.Color = RGB(0, 176, 80)
            End If
        End With

    Next r

End Sub

```


## 三、怎么用（很快）

1. `Alt + F11` → VBA 编辑器
    
2. `插入 → 模块`
    
3. 粘贴代码
    
4. `Alt + F8` → 运行 `ColorByLogic_AM_AN_AO_AP`

## 四、可选升级（实战常用）

### ✔ 数据一变，颜色自动刷新
```
Private Sub Worksheet_Change(ByVal Target As Range)

    If Intersect(Target, Range("X:AK")) Is Nothing Then Exit Sub
    
    Application.EnableEvents = False
    Call ColorByLogic_AM_AN_AO_AP
    Application.EnableEvents = True

End Sub

```


👉 放在 **工作表代码页**，不是模块。
