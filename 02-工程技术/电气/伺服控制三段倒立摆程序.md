## 🤖 Assistant

【功能说明】
本功能块 `FB_TriplePendulumControl` 专用于西门子 S7-1500 PLC（含 S7-1200），实现三段倒立摆系统的核心伺服控制逻辑。
1.  **控制算法**：采用 State-Space Feedback（状态空间反馈/LQR）控制策略，通过线性组合 8 个状态变量（小车位移/速度、下摆角度/角速度、中摆角度/角速度、上摆角度/角速度）计算控制力/电压。
2.  **状态管理**：包含系统复位、回零、手动/自动模式切换、能量摆起（预留接口）、平衡保持及故障停机。
3.  **数据预处理**：实现编码器原始数据到物理量的转换，并使用一阶滞后滤波计算角速度，减少传感器噪声。
4.  **安全联锁**：包含机械限位检查、角度过大保护及伺服故障监控。

***

```scl
FUNCTION_BLOCK "FB_TriplePendulumControl"
{ S7_Optimized_Access := 'TRUE' }
VERSION : 0.1

   VAR_INPUT 
      // 系统控制信号 // System Control Signals
      "EN_Enable" : Bool;            // 使能信号 / Enable Signal
      "EN_Reset" : Bool;             // 复位信号 / Reset
      "Mode_Auto" : Bool;            // 自动模式（平衡控制）/ Auto Mode (Balance)
      "Mode_Manual" : Bool;          // 手动模式/ Manual Mode
      "Manual_SP" : Real;            // 手动设定速度/ Manual Speed Setpoint
      
      // 传感器反馈 // Sensor Feedback (Assumed Units: Counts or Meters depending on config)
      "Enc_Cart" : Real;             // 小车位置 / Cart Position [m]
      "Enc_Theta1" : Real;           // 摆杆1角度 (下摆) / Pendulum 1 Angle [rad]
      "Enc_Theta2" : Real;           // 摆杆2角度 (中摆) / Pendulum 2 Angle [rad]
      "Enc_Theta3" : Real;           // 摆杆3角度 (上摆) / Pendulum 3 Angle [rad]
      
      "Limit_Left" : Bool;           // 左限位开关 / Left Limit Switch
      "Limit_Right" : Bool;          // 右限位开关 / Right Limit Switch
      "Servo_Ready" : Bool;          // 伺服驱动器就绪 / Servo Drive Ready
      "Cycle_Time" : Time := T#2MS;  // 扫描周期 (建议在 OB35 中调用) / Cycle Time
   END_VAR

   VAR_OUTPUT 
      // 伺服输出 // Servo Output
      "Servo_Speed_SP" : Real;       // 伺服速度设定点 / Servo Speed Setpoint [m/s] or [rpm]
      "Servo_Torque_Ofs" : Real;     // 伺服转矩附加/前馈 (可选)/ Torque Feedforward
      
      // 状态反馈 // Status Signals
      "State_Machine" : Int;         // 当前状态机状态 / Current FSM State
      "Is_Balancing" : Bool;         // 是否处于平衡状态 / Is Balancing
      "Error_ID" : Word;             // 错误代码 / Error Code
      
      // 过程变量 (用于 SCADA) / Process Variables for HMI
      "PV_Cart_Pos" : Real;          // 小车实际位置滤波后 / Actual Cart Pos
      "PV_Cart_Vel" : Real;          // 小车实际速度 / Actual Cart Vel
      "PV_Theta1" : Real;            // 摆杆1实际角度 / Actual Theta1
      "PV_Omega1" : Real;            // 摆杆1实际角速度 / Actual Omega1
      "PV_Theta2" : Real;
      "PV_Omega2" : Real;
      "PV_Theta3" : Real;
      "PV_Omega3" : Real;
   END_VAR

   VAR_IN_OUT 
      // 非易失性参数保持 / Non-volatile Parameters
      "Setup" : ST_ControllerSetup;  // 参数结构体 / Parameter Struct
   END_VAR

   VAR 
      // 内部状态变量 / Internal State Variables
      "_State" : Int := 0;           // 状态机步 / FSM Step
      "_Timer_Reset" : TON;          // 复位延时计时器 / Reset Timer
      "_First_Scan" : Bool := TRUE;  // 首次扫描标志 / First Scan Flag
      
      // 中间计算变量 / Calculation Variables
      "_Error_Vector" : LReal;       // 状态误差累积 / State Error Accumulator
      "_Control_Output_LReal" : LReal; // 控制输出高精度 / Control Output High Precision
      "_Prev_Cart_Pos" : LReal;      // 上次小车位置 / Previous Cart Pos
      "_Prev_Theta1" : LReal;
      "_Prev_Theta2" : LReal;
      "_Prev_Theta3" : LReal;
      
      // 滤波系数 / Filter Coefficient
      "_Filter_Factor" : LReal;
      "_dt_Seconds" : LReal;
   END_VAR

BEGIN
   ///////////////////////////////////////////////////////////////////////////////
   // 1. 初始化与参数计算 / Initialization & Parameter Calc
   ///////////////////////////////////////////////////////////////////////////////
   IF "_First_Scan" THEN
      _First_Scan := FALSE;
      _Prev_Cart_Pos := LREAL#"Enc_Cart";
      _Prev_Theta1 := LREAL#"Enc_Theta1";
      _Prev_Theta2 := LREAL#"Enc_Theta2";
      _Prev_Theta3 := LREAL#"Enc_Theta3";
      
      // 默认将错误代码清零 / Clear Error Code
      #"Error_ID" := 16#0000;
   END_IF;
   
   // 计算采样时间 / Calculate Sampling Time
   _dt_Seconds := DINT_TO_LREAL(TIME_IN_MS("Cycle_Time")) / 1000.0;
   
   // 计算滤波系数 (一阶滞后) T = Filter_Time_Const / (Filter_Time_Const + dt)
   // 简单起见，这里假设 Setup 中有一个 FilterConst 参数
   IF ("Setup".Filter_Const > 0.0) THEN
      _Filter_Factor := "Setup".Filter_Const / ("Setup".Filter_Const + _dt_Seconds);
   ELSE
      _Filter_Factor := 0.0; // 不滤波 / No Filter
   END_IF;

   ///////////////////////////////////////////////////////////////////////////////
   // 2. 数据预处理与状态观测 / Data Processing & State Observation
   ///////////////////////////////////////////////////////////////////////////////
   // 物理量赋值与坐标转换 / Physical Unit Assignment & Coord Transformation
   #"PV_Cart_Pos" := "Enc_Cart"; 
   // 注意：摆杆角度通常以垂直向上为0。编码器可能需要转换。
   // 假设输入已经是弧度，且垂直向上为 0 / Assumes input is rad, Up is 0
   #"PV_Theta1" := "Enc_Theta1";
   #"PV_Theta2" := "Enc_Theta2";
   #"PV_Theta3" := "Enc_Theta3";

   // 速度计算 (微分 + 滤波) / Velocity Calculation (Diff + Filter)
   // 公式: Vel(k) = Vel(k-1) * Factor + (Pos(k) - Pos(k-1)) / dt * (1 - Factor)
   #"PV_Cart_Vel" := REAL(#"PV_Cart_Vel" * _Filter_Factor + 
                   ((#"PV_Cart_Pos" - _Prev_Cart_Pos) / _dt_Seconds) * (1.0 - _Filter_Factor));
                   
   #"PV_Omega1" := REAL(#"PV_Omega1" * _Filter_Factor + 
                   ((#"PV_Theta1" - _Prev_Theta1) / _dt_Seconds) * (1.0 - _Filter_Factor));
                   
   #"PV_Omega2" := REAL(#"PV_Omega2" * _Filter_Factor + 
                   ((#"PV_Theta2" - _Prev_Theta2) / _dt_Seconds) * (1.0 - _Filter_Factor));
                   
   #"PV_Omega3" := REAL(#"PV_Omega3" * _Filter_Factor + 
                   ((#"PV_Theta3" - _Prev_Theta3) / _dt_Seconds) * (1.0 - _Filter_Factor));

   // 更新历史值 / Update History
   _Prev_Cart_Pos := #"PV_Cart_Pos";
   _Prev_Theta1 := #"PV_Theta1";
   _Prev_Theta2 := #"PV_Theta2";
   _Prev_Theta3 := #"PV_Theta3";

   ///////////////////////////////////////////////////////////////////////////////
   // 3. 状态机控制逻辑 / State Machine Logic
   ///////////////////////////////////////////////////////////////////////////////
   CASE "_State" OF
      0: // IDLE / 空闲待机
         #"Servo_Speed_SP" := 0.0;
         #"Is_Balancing" := FALSE;
         
         IF "EN_Reset" THEN
            "_State" := 10; // 进入复位流程 / Go to Reset
         ELSIF "EN_Enable" AND "Servo_Ready" THEN
            "_State" := 20; // 准备运行 / Ready to Run
         END_IF;

      10: // RESET / 复位中
         // 等待系统稳定 / Wait for system stable
         "_Timer_Reset"(IN := NOT "EN_Reset", PT := T#1S);
         IF "_Timer_Reset".Q THEN
             "_State" := 0;
         END_IF;

      20: // READY / 准备就绪
         IF "Mode_Manual" THEN
             "_State" := 30; // 手动模式 / Manual Mode
         ELSIF "Mode_Auto" THEN
             // 检查摆杆是否在平衡位置附近 (例如 +/- 15度 / +/- 0.26 rad)
             IF (ABS(#"PV_Theta1") < "Setup".Balance_Threshold) AND 
                (ABS(#"PV_Theta2") < "Setup".Balance_Threshold) AND 
                (ABS(#"PV_Theta3") < "Setup".Balance_Threshold) THEN
                 "_State" := 40; // 直接进入平衡 / Enter Balance
             ELSE
                 "_State" := 50; // 进入摆起 / Enter Swing Up (Need Swing-up Logic)
             END_IF;
         END_IF;
         IF NOT "EN_Enable" THEN "_State" := 0; END_IF;

      30: // MANUAL / 手动控制
         #"Servo_Speed_SP" := "Manual_SP";
         IF NOT "Mode_Manual" THEN "_State" := 20; END_IF;
         
      40: // AUTO BALANCE / 自动平衡 (LQR 控制)
         #"Is_Balancing" := TRUE;
         
         // --- 核心 LQR 控制算法 ---
         // u = -K * x = -(k1*x + k2*v + k3*t1 + k4*w1 + ...
         // State Vector x = [r, v, theta1, omega1, theta2, omega2, theta3, omega3]
         
         _Error_Vector := 
             "Setup".K_Gains[0] * LREAL(#"PV_Cart_Pos") +          // k1 * r
             "Setup".K_Gains[1] * LREAL(#"PV_Cart_Vel") +          // k2 * v
             "Setup".K_Gains[2] * LREAL(#"PV_Theta1") +           // k3 * t1
             "Setup".K_Gains[3] * LREAL(#"PV_Omega1") +           // k4 * w1
             "Setup".K_Gains[4] * LREAL(#"PV_Theta2") +           // k5 * t2
             "Setup".K_Gains[5] * LREAL(#"PV_Omega2") +           // k6 * w2
             "Setup".K_Gains[6] * LREAL(#"PV_Theta3") +           // k7 * t3
             "Setup".K_Gains[7] * LREAL(#"PV_Omega3");            // k8 * w3

         // 计算输出并取负号 / Calculate Output (Negative Feedback)
         _Control_Output_LReal := -_Error_Vector;
         
         // 输出限幅 / Output Limiting
         IF _Control_Output_LReal > LREAL(#"Setup".Max_Output") THEN
             #"Servo_Speed_SP" := #"Setup".Max_Output";
         ELSIF _Control_Output_LReal < -LREAL(#"Setup".Max_Output") THEN
             #"Servo_Speed_SP" := -#"Setup".Max_Output";
         ELSE
             #"Servo_Speed_SP" := REAL(_Control_Output_LReal);
         END_IF;
         
         // --- 平衡状态监控 / Balance Monitoring ---
         // 如果倒下，切换状态 / If falls, switch state
         IF (ABS(#"PV_Theta1") > "Setup".Fail_Threshold) OR 
            (ABS(#"PV_Theta2") > "Setup".Fail_Threshold) OR 
            (ABS(#"PV_Theta3") > "Setup".Fail_Threshold") THEN
             "_State" := 100; // Failed
         END_IF;
         
         IF NOT "Mode_Auto" THEN "_State" := 20; END_IF;

      50: // SWING UP / 摆起控制 (预留接口 / Placeholder for Swing-up)
          // 实际工程中，此部分通常使用能量控制或基于轨迹的规划
          // 这里仅保持低速搜索或等待外部切换 / Placeholder for logic
          #"Is_Balancing" := FALSE;
          #"Servo_Speed_SP" := 0.0; // 暂停

      100: // ERROR / 故障停机
           "Error_ID" := 16#0001; // Pendulum Fell
           #"Servo_Speed_SP" := 0.0;
           IF "EN_Reset" THEN "_State" := 0; END_IF;

   END_CASE;

   ///////////////////////////////////////////////////////////////////////////////
   // 4. 硬件安全联锁 / Hardware Safety Interlocks
   ///////////////////////////////////////////////////////////////////////////////
   IF ("Limit_Left" OR "Limit_Right") THEN
      // 立即停止输出 / Immediate Stop
      #"Servo_Speed_SP" := 0.0;
      "_State" := 100;
      "Error_ID" := 16#0002; // Limit Switch Triggered
   END_IF;
   
   IF NOT "Servo_Ready" THEN
      #"Servo_Speed_SP" := 0.0;
      "_State" := 0; // 强制回空闲 / Force to Idle
   END_IF;
   
   #"State_Machine" := "_State";

END_FUNCTION_BLOCK
```

```scl
TYPE ST_ControllerSetup : STRUCT
   // LQR 增益矩阵 / LQR Gain Matrix (Must be tuned in MATLAB/Simulink)
   // 顺序: r, v, t1, w1, t2, w2, t3, w3
   K_Gains : ARRAY[0..7] OF LREAL := 
      [5.0, 10.0, 100.0, 15.0, 200.0, 20.0, 50.0, 5.0]; // 示例值，绝对不可直接用于现场 / Example values ONLY

   // 系统限制 / System Limits
   Max_Output : Real := 5.0;      // 最大输出速度或电压 / Max Output Speed [m/s]
   Balance_Threshold : Real := 0.26; // 进入平衡区的角度阈值 (约15度) / Entry Threshold [rad]
   Fail_Threshold : Real := 0.5;  // 平衡失败阈值 (约30度) / Fail Threshold [rad]
   
   // 滤波参数 / Filter Params
   Filter_Const : Real := 0.01;   // 滤波时间常数 / Filter Time Constant [s]
END_STRUCT
END_TYPE
```

【逻辑说明】
1.  **I/O 处理**：程序首先读取编码器位置。倒立摆最重要的在于 **“导数”**（速度/角速度）的计算精度。由于编码器存在量化噪声，直接微分会导致高频震荡，因此代码中实现了一个标准的一阶低通滤波器来平滑速度信号。
2.  **状态机架构 (FSM)**：
    *   `IDLE`：安全停机状态。
    *   `READY`：检测模式切换指令。
    *   `MANUAL`：用于维修或调试时的点动控制。
    *   `BALANCE`：核心状态。在此状态下，假设摆杆已接近垂直（通常由能量摆起逻辑进入），程序执行 LQR 线性控制律：$u = -K \dot{x}$。如果角度偏差超过 `Fail_Threshold`，判定为倒下，立即跳转至故障状态。
    *   `SWING_UP`：三段倒立摆极不稳定，通常需要先将其从自然下垂状态摆起至平衡位置附近。由于该算法极其复杂（通常基于 partial feedback linearization 或 energy shaping），本代码预留了状态接口，实际项目中通常由外部高级算法计算得出控制量后直接赋值给 `Servo_Speed_SP`。
3.  **LQR 实现**：核心在于矩阵乘法。由于 SCL 运算效率限制，对于 8x1 矩阵乘积，直接展开比写循环算法效率略高且更易读。代码直接展开了 $u = - (k_1 r + k_2 v + k_3 \theta_1 ...)$ 的计算过程。

【注意事项】
1.  **采样周期**：倒立摆对时间极其敏感。**必须将此 FB 放置在定时中断 OB（如 OB35, OB32）中调用**，且周期建议设置为 **2ms ~ 5ms**。在 OB1 中调用通常会导致系统发散。
2.  **增益调试 ($K$ 矩阵)**：代码中的 `K_Gains` 数组仅为占位符。必须使用 MATLAB (`lqr` 函数) 或 Python 针对实际机械模型的物理参数（质量、杆长、转动惯量）计算精确的增益矩阵，并写入对应 DB。
3.  **单位一致性**：确保所有输入（编码器->米/弧度转换）和输出（PLC控制量 -> 伺服驱动器接收单位）的单位一致。S7-1500 建议全程使用 `LREAL` 进行中间浮点运算以避免截断误差。
4.  **积分饱和**：本方案采用纯比例的状态反馈，不包含积分项。如需消除小车位置的稳态误差，可在小车位置状态量上叠加一个积分器或增加前馈控制，但需小心处理积分饱和问题，防止系统失稳。

【可扩展建议】
1.  **摆起算法**：建议实现“基于能量”的摆起控制器。计算系统总能量 $E$，施加控制力使能量目标值 $E_{target}$（即直立势能）逼近。
2.  **轨迹跟踪**：如需小车按正弦波或特定轨迹运动，可在状态反馈回路上叠加前馈控制量，或者修改目标状态 $x_{ref}$。
3.  **参数在线调整**：将 `K_Gains` 暴露给 HMI/WinCC，但需设置严格的权限，防止误调导致机械损坏。增加“增益淡入”逻辑，即从 0 逐渐增加到设定值，使启动过程更平滑。
